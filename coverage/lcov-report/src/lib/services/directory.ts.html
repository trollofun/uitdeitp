/**
 * Datele directorului public de stații.
 *
 * Citirea trece prin clientul de serviciu, nu prin RLS: fișele sunt publice
 * prin definiție, iar o politică `USING (public_listed)` ar însemna că fiecare
 * vizitator anonim primește totuși un rând întreg din `kiosk_stations` — cu
 * chei NotifyHub, praguri de credit și șabloane. Aici alegem coloanele.
 */
 
import { createServiceClient } from '@/lib/supabase/service';
import { ROMANIAN_COUNTIES } from '@/lib/services/plate';
import { parseWorkingHours } from '@/lib/services/booking/slots';
 
/** Coloanele care au voie să ajungă la un vizitator. Lista albă, nu excluderi. */
const PUBLIC_COLUMNS =
  'slug, name, city, county_code, station_address, station_phone, public_description, pricing, working_hours, booking_enabled, latitude, longitude';
 
export interface PublicStation {
  slug: string;
  name: string;
  city: string | null;
  county_code: string | null;
  county_name: string | null;
  station_address: string | null;
  station_phone: string | null;
  public_description: string | null;
  pricing: Array<{ label: string; price_lei: number }>;
  working_hours: Record<string, [string, string][]>;
  booking_enabled: boolean;
}
 
const DAY_NAMES = ['', 'Luni', 'Marți', 'Miercuri', 'Joi', 'Vineri', 'Sâmbătă', 'Duminică'];
 
/** Programul, în formă citibilă: „Luni-Vineri 08:00-16:00". */
export function formatWorkingHours(raw: unknown): string[] {
  const hours = parseWorkingHours(raw);
  const lines: string[] = [];
 
  // Zilele consecutive cu același program se grupează. Șapte rânduri identice
  // pe o fișă publică sunt zgomot, nu informație.
  let runStart: number | null = null;
  let runValue = '';
 
  const valueOf = (day: number) =>
    (hours[String(day)] ?? []).map(([a, b]) => `${a}-${b}`).join(', ');
 
  for (let day = 1; day <= 8; day += 1) {
    const value = day <= 7 ? valueOf(day) : '';
 
    if (value !== runValue) {
      if (runStart !== null && runValue) {
        const label =
          runStart === day - 1
            ? DAY_NAMES[runStart]
            : `${DAY_NAMES[runStart]}–${DAY_NAMES[day - 1]}`;
        lines.push(`${label}: ${runValue}`);
      }
      runStart = day;
      runValue = value;
    }
  }
 
  return lines;
}
 
function toPublic(row: Record<string, unknown>): PublicStation {
  const code = (row.county_code as string | null) ?? null;
 
  return {
    slug: row.slug as string,
    name: row.name as string,
    city: (row.city as string | null) ?? null,
    county_code: code,
    county_name: code ? (ROMANIAN_COUNTIES[code] ?? null) : null,
    station_address: (row.station_address as string | null) ?? null,
    station_phone: (row.station_phone as string | null) ?? null,
    public_description: (row.public_description as string | null) ?? null,
    pricing: Array.isArray(row.pricing)
      ? (row.pricing as Array<{ label: string; price_lei: number }>)
      : [],
    working_hours: parseWorkingHours(row.working_hours),
    booking_enabled: Boolean(row.booking_enabled),
  };
}
 
export async function listPublicStations(countyCode?: string): Promise<PublicStation[]> {
  let query = createServiceClient()
    .from('kiosk_stations')
    .select(PUBLIC_COLUMNS)
    .eq('public_listed', true)
    .eq('is_active', true)
    .order('city')
    .order('name');
 
  if (countyCode) query = query.eq('county_code', countyCode.toUpperCase());
 
  const { data, error } = await query;
 
  if (error) {
    console.warn('[Directory] listing failed', { code: error.code });
    return [];
  }
 
  return (data ?? []).map((row) => toPublic(row as Record<string, unknown>));
}
 
export async function getPublicStation(slug: string): Promise<PublicStation | null> {
  const { data } = await createServiceClient()
    .from('kiosk_stations')
    .select(PUBLIC_COLUMNS)
    .eq('slug', slug)
    .eq('public_listed', true)
    .eq('is_active', true)
    .maybeSingle();
 
  return data ? toPublic(data as Record<string, unknown>) : null;
}
 
/** Județele care au cel puțin o stație listată — pentru indexul directorului. */
export async function listCountiesWithStations(): Promise<
  Array<{ code: string; name: string; count: number }>
> {
  const stations = await listPublicStations();
  const counts = new Map<string, number>();
 
  for (const station of stations) {
    if (!station.county_code) continue;
    counts.set(station.county_code, (counts.get(station.county_code) ?? 0) + 1);
  }
 
  return [...counts.entries()]
    .map(([code, count]) => ({ code, name: ROMANIAN_COUNTIES[code] ?? code, count }))
    .sort((a, b) => a.name.localeCompare(b.name, 'ro'));
}
